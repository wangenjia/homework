/* ========================================================= 前端逻辑（Commit 1：棋盘与箭头渲染） ========================================================= */ const DIRS = { up: { dr: -1, dc: 0, glyph: '↑' }, down: { dr: 1, dc: 0, glyph: '↓' }, left: { dr: 0, dc: -1, glyph: '←' }, right: { dr: 0, dc: 1, glyph: '→' }, }; const boardEl = document.getElementById('board'); const overlayEl = document.getElementById('overlay'); const hudLevel = document.getElementById('hud-level'); const hudArrows = document.getElementById('hud-arrows'); const hudMistake = document.getElementById('hud-mistakes'); let game = { state: null, cell: 0 }; let busy = false; /* ---------- 通用工具 ---------- */ function showScreen(name) { document.querySelectorAll('.screen').forEach(s => s.classList.remove('active')); document.getElementById('screen-' + name).classList.add('active'); } function showDialog({ icon, title, msg, primary, secondary }) { document.getElementById('dlg-icon').textContent = icon || ''; document.getElementById('dlg-title').textContent = title; document.getElementById('dlg-msg').innerHTML = msg || ''; const pBtn = document.getElementById('dlg-primary'); const sBtn = document.getElementById('dlg-secondary'); pBtn.textContent = primary.label; pBtn.onclick = primary.fn; if (secondary) { sBtn.style.display = ''; sBtn.textContent = secondary.label; sBtn.onclick = secondary.fn; } else { sBtn.style.display = 'none'; } overlayEl.classList.add('show'); } function computeCell(lv) { const availW = Math.min(window.innerWidth - 40, 460); const availH = window.innerHeight - 250; const byW = availW / lv.cols; const byH = Math.max(availH, 200) / lv.rows; return Math.max(38, Math.min(Math.floor(Math.min(byW, byH)), 78)); } /* ---------- 与后端通信 ---------- */ async function api(url, body) { const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body || {}), }); return res.json(); } /* ---------- 渲染棋盘 ---------- */ function renderBoard(state) { const cell = computeCell(state); game.cell = cell; boardEl.style.width = cell * state.cols + 'px'; boardEl.style.height = cell * state.rows + 'px'; boardEl.style.setProperty('--cell', cell + 'px'); boardEl.innerHTML = ''; state.arrows.forEach(a => addArrowEl(a.r, a.c, a.dir)); } function addArrowEl(r, c, dir) { const cell = game.cell; const el = document.createElement('div'); el.className = 'arrow ' + dir; el.style.left = c * cell + 'px'; el.style.top = r * cell + 'px'; el.style.width = cell + 'px'; el.style.height = cell + 'px'; const face = document.createElement('div'); face.className = 'face'; face.textContent = DIRS[dir].glyph; el.appendChild(face); // 注意：Commit 1 不绑定点击事件 boardEl.appendChild(el); } /* ---------- HUD 与状态 ---------- */ function updateHUD(state) { hudLevel.textContent = (state.level_index + 1) + ' / ' + state.level_total; hudArrows.textContent = state.arrows_left; hudMistake.textContent = state.mistakes_left; } /* ---------------- 点击箭头 ---------------- */ async function onArrowClick(r, c, el, dir) { if (busy || !game || !game.state || game.state.locked) return; busy = true; const resp = await api('/api/click', { r, c }); if (!resp.ok) { busy = false; return; } if (resp.result === 'blocked') { handleBlocked(el, resp); } else if (resp.result === 'flew') { handleFlew(el, resp); } } /* 被阻挡：抖动 / function handleBlocked(el, resp) { el.classList.remove('shake'); void el.offsetWidth; el.classList.add('shake'); } /* 飞出：/ function handleFlew(el, resp) { const cell = game.cell; const { dr, dc } = DIRS[resp.arrow.dir]; const rows = game.state.rows, cols = game.state.cols; let dist; if (dc > 0) dist = (cols - resp.arrow.c + 1) * cell; else if (dc < 0) dist = (resp.arrow.c + 2) * cell; else if (dr > 0) dist = (rows - resp.arrow.r + 1) * cell; else dist = (resp.arrow.r + 2) * cell; el.classList.add('flying'); el.style.pointerEvents = 'none'; requestAnimationFrame(() => { el.style.transform = `translate(${dc * dist}px, ${dr * dist}px)`; el.style.opacity = '0'; }); hudArrows.textContent = resp.arrows_left; setTimeout(() => { el.remove(); game.state = resp.state; busy = false; if (resp.cleared) { const isLast = resp.state.level_index + 1 >= resp.state.level_total; setTimeout(() => { if (!isLast) { showDialog({ icon: '🎉', title: '通关成功', msg: `第 ${resp.state.level_index + 1} 关完成！
剩余失误 ${resp.state.mistakes_left} 次。`, primary: { label: '进入下一关', fn: () => nextLevel() }, secondary: { label: '重玩本关', fn: () => restartLevel() }, }); } } } } /* ---------------- 飘字 ---------------- */ function floatMissText(r, c) { const cell = game.cell; const el = document.createElement('div'); el.className = 'float-miss'; el.textContent = '-1'; el.style.left = c * cell + 'px'; el.style.top = r * cell + 'px'; el.style.width = cell + 'px'; el.style.height = cell + 'px'; boardEl.appendChild(el); setTimeout(() => el.remove(), 820); } /* ---------------- 生命周期 ---------------- */ async function startLevel(index) { hideDialog(); const resp = await api('/api/start', { index }); if (!resp.ok) return; showScreen('game'); applyState(resp.state); } /* ---------- 事件绑定 ---------- */ document.getElementById('btn-start').addEventListener('click', () => startLevel(0)); // 重新开始按钮暂不绑定 /* ---------- 初始 ---------- */ showScreen('start');
